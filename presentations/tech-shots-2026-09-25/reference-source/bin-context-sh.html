36# Canonical path for a NEW checkpoint. Caller writes content to it.
37# Args: [label]
38context_save_path() {
39 local label="${1:-}"
40 local branch dir ts slug
41 branch="$(_context_branch)"
42 dir="$(lintel_sessions_dir 2>/dev/null)" || dir="$LINTEL_HOME/sessions"
43 dir="$dir/$branch"
44 mkdir -p "$dir" 2>/dev/null || true
45 ts=$(date +"%Y%m%d-%H%M%S")
46 slug="$(_context_repo_slug)"
47 # slugify the label: separators/spaces → '-', keep it filename-safe
48 label=$(printf '%s' "$label" | tr ' /\\' '---' | tr -cd 'A-Za-z0-9._-')
49 if [ -n "$label" ]; then
50 printf '%s/%s-%s-%s-context-save.md' "$dir" "$ts" "$slug" "$label"
51 else
52 printf '%s/%s-%s-context-save.md' "$dir" "$ts" "$slug"
53 fi
54}

56# All checkpoints for a branch, newest first. Includes the legacy global dir
57# (read-only) during the grace window.
58# Args: [branch] (default: current)
59context_list() {
60 local branch="${1:-$(_context_branch)}"
61 local newdir
62 newdir="$(lintel_sessions_dir 2>/dev/null)" || newdir=""
63 {
64 [ -n "$newdir" ] && [ -d "$newdir/$branch" ] && \
65 find "$newdir/$branch" -maxdepth 1 -name '*-context-save.md' 2>/dev/null
66 # Legacy dir only when it is NOT what lintel_sessions_dir already resolved
67 # to (un-migrated repos resolve THERE — listing it twice double-counts).
68 if [ "$newdir" != "$LINTEL_HOME/sessions" ] && [ -d "$LINTEL_HOME/sessions/$branch" ]; then
69 find "$LINTEL_HOME/sessions/$branch" -maxdepth 1 -name '*-context-save.md' 2>/dev/null # legacy-fallback-ok
70 fi
71 } | while IFS= read -r f; do
72 # prefix with basename for chronological sort (timestamps lead the name)
73 printf '%s\t%s\n' "$(basename "$f")" "$f"
74 done | sort -r | cut -f2 | awk '!seen[$0]++'
