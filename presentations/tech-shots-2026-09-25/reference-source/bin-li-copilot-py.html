311def atomic_write(path: Path, data: bytes) -> None:
312 path.parent.mkdir(parents=True, exist_ok=True)
313 fd, tmp = tempfile.mkstemp(prefix=".lintel-", dir=path.parent)
314 try:
315 with os.fdopen(fd, "wb") as handle:
316 handle.write(data)
317 os.replace(tmp, path)
318 finally:
319 if os.path.exists(tmp):
