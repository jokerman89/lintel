250# ─── Cache management ──────────────────────────────────────────────────────
251_prime_cache_for_session() {
252 # Reads active pack, validates, copies merged manifest to cache.
253 # Run once per session. Subsequent resolve_pack_field calls hit cache.
254 if [ -f "$PACK_CACHE_FILE" ]; then
255 # mtime invalidation: a pack.yaml edited AFTER the cache was primed must not
256 # stay invisible until a manual clear_pack_cache. The active-pack POINTER is
257 # deliberately not checked — mid-session pack-switch keeps the cached value
258 # per the §2.4 contract (fallback-harness scenario 7); a deleted pack keeps
259 # the cache too (scenario 8: _pack_dir fails → manifest not checked).
260 local _name _p _pd _stale=0
261 _name=$(get_active_pack_name)
262 for _p in $(_resolve_extends_chain "$_name"); do
263 _pd=$(_pack_dir "$_p" 2>/dev/null) || continue
264 [ "$_pd/pack.yaml" -nt "$PACK_CACHE_FILE" ] && _stale=1
265 done
266 if [ "$_stale" = 0 ]; then
267 return 0 # already primed and fresh
268 fi
269 _resolver_audit cache_invalidated "pack=$_name reason=manifest-newer-than-cache session=$LINTEL_SESSION_ID"
270 rm -f "$PACK_CACHE_FILE" 2>/dev/null || true
271 fi
272
273 local name
274 name=$(get_active_pack_name)
275
276 if ! validate_pack "$name"; then
277 # Failure mode 4: internal bug or invalid active pack → fall back to _default
278 _resolver_warn "active pack '$name' invalid; falling back to _default"
279 name="_default"
280 if ! validate_pack "$name"; then
281 _resolver_fail "_default pack ALSO invalid — repo is broken"
282 return 1
283 fi
