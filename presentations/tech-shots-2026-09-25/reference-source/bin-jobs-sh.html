56# Generate a job-id: <workflow>-<YYYYMMDD-HHMM>-<short-hash>
57job_id() {

59 local stamp
60 stamp=$(date +"%Y%m%d-%H%M")

62 hash=$(printf '%s' "$RANDOM-$$-$(_jobs_epoch_now)" | shasum 2>/dev/null | cut -c1-6 || \

116# ── Per-step contracts (design §3.4 / docs/concepts/jobs-system.md "Mechanics") ──
117#
118# A <step-spec> is a single pipe-delimited record describing one WBS leaf/phase:
119#
120# NAME|consumes=a.md,b.md|produces=c.md,d.md|status=PENDING|blocked_until=PLAN.status == DONE
121#
122# - field 1 (positional) is the step NAME. For flat/phased plans this is a
123# phase token (PLAN, BUILD); for tree plans it is a WBS node-path (1.1.a).
124# - consumes / produces are comma-separated artifact lists (rendered as YAML
125# inline sequences). Omit → [].
126# - status defaults to PENDING.
127# - blocked_until is a predicate over OTHER steps' status (grammar below).
128#
129# Field order after NAME is free; unknown keys are ignored.
130#
131# blocked_until predicate grammar (kept deliberately small — see job_can_start):
132# <predicate> := <clause> ( "&&" <clause> )*
133# <clause> := <STEP>.status == DONE
134# i.e. a conjunction of "this named step has reached DONE" tests. Empty/absent
135# means "never blocked". No OR, no negation, no other status values — by design.
136
137# Render one step-spec to indented YAML on stdout.
138_jobs_render_step() {
139 local spec="$1"
140 local name consumes produces status blocked field key val
141 name=""; consumes=""; produces=""; status="PENDING"; blocked=""
142
143 # First pipe-field is the name; the rest are key=value.
144 name="${spec%%|*}"
145 local rest="${spec#"$name"}"
146 rest="${rest#|}"
147
148 # Split remaining fields on '|'
149 local IFS_SAVE="$IFS"
150 IFS='|'
151 # shellcheck disable=SC2206
152 local fields=($rest)
153 IFS="$IFS_SAVE"
154 for field in "${fields[@]}"; do
155 [ -n "$field" ] || continue
156 key="${field%%=*}"
157 val="${field#*=}"
158 case "$key" in
159 consumes) consumes="$val" ;;
160 produces) produces="$val" ;;
161 status) status="$val" ;;
162 blocked_until) blocked="$val" ;;
163 esac
164 done

166 # Comma-list → YAML inline sequence "[a, b]"
167 _jobs_yaml_seq() {
168 local csv="$1"
169 [ -n "$csv" ] || { printf '[]'; return; }
170 local out="" item
171 local IFS=','
172 # shellcheck disable=SC2206
173 local items=($csv)
174 for item in "${items[@]}"; do
175 item="${item# }"; item="${item%% }"
176 [ -n "$item" ] || continue
177 [ -n "$out" ] && out="$out, $item" || out="$item"
178 done
179 printf '[%s]' "$out"
180 }
181
182 printf ' - name: %s\n' "$name"
183 printf ' consumes: %s\n' "$(_jobs_yaml_seq "$consumes")"
184 printf ' produces: %s\n' "$(_jobs_yaml_seq "$produces")"
185 printf ' status: %s\n' "$status"
186 if [ -n "$blocked" ]; then
187 printf ' blocked_until: %s\n' "$blocked"
188 fi
189}

194job_set_steps() {

207 # Reprint job.yaml up to (not including) the `steps:` line, then re-emit steps.
208 # The steps block is always the trailing block (job_create writes it last),
209 # so dropping from `^steps:` to EOF and re-appending is safe and idempotent.
210 awk '
211 /^steps:/ { found=1 }
212 !found { print }
213 ' "$dir/job.yaml" > "$dir/job.yaml.tmp"
214
215 if [ -z "$body" ]; then
216 printf 'steps: []\n' >> "$dir/job.yaml.tmp"
217 else
218 printf 'steps:\n' >> "$dir/job.yaml.tmp"
219 printf '%s\n' "$body" >> "$dir/job.yaml.tmp"
220 fi

288# Archive a job + apply cleanup
289# Args: <id> <result> result in: DONE / ABORTED / FAILED
290job_archive() {
291 local id="$1"
292 local result="${2:-DONE}"
293 local dir="$LINTEL_JOBS_DIR/$id"
294 [ -d "$dir" ] || return 1
295
296 local date_stamp
297 date_stamp=$(date +"%Y-%m-%d")
298 local archive_dir="$LINTEL_JOBS_ARCHIVE/$date_stamp"
299 mkdir -p "$archive_dir" 2>/dev/null || true
300
301 # Apply cleanup-policy: discard scratch/*
302 if [ -d "$dir/scratch" ]; then
303 rm -rf "$dir/scratch" 2>/dev/null || true
304 fi
305
306 # Mark final status + ts in job.yaml
307 local ts
308 ts=$(_jobs_iso_now)
309 awk -v ts="$ts" -v result="$result" '
310 BEGIN { updated_lt=0; updated_st=0; appended_arc=0 }
311 /^last_touched:/ && !updated_lt { print "last_touched: " ts; updated_lt=1; next }
312 /^status:/ && !updated_st { print "status: " result; updated_st=1; next }
313 { print }
314 END { print "archived_at: " ts }
315 ' "$dir/job.yaml" > "$dir/job.yaml.tmp" && mv "$dir/job.yaml.tmp" "$dir/job.yaml"
316
317 # Move to archive (mv across paths)
318 mv "$dir" "$archive_dir/" 2>/dev/null || true
319
320 audit_log "jobs" "job_end" "job_id=$id" "result=$result"
321
322 regenerate_active

376# ── cross-repo registry (v5) ─────────────────────────────────────────────────
377# ~/.lintel/jobs/_active.md holds one line per open job across ALL scopes.
378# Each line carries an HTML-comment marker with the owning scope (a repo root,
379# or "global" for legacy ~/.lintel/jobs data); syncing = drop this scope's
380# lines, re-append the current ones, preserve everyone else's.
381# Args: <scope-tag> (repo root path, or "global")
382_registry_sync() {
383 local scope_tag="${1:-$_JOBS_REPO_ROOT}"
384 local reg="$LINTEL_JOBS_REGISTRY"
385 mkdir -p "$(dirname "$reg")" 2>/dev/null || true
386
387 local marker="<!-- repo:$scope_tag -->"
388 local tmp="$reg.tmp.$$"
389 {
390 if [ -f "$reg" ] && grep -q '^# Lintel jobs registry' "$reg" 2>/dev/null; then
391 grep -vF "$marker" "$reg" | grep -v '^_No open jobs' || true
392 else
393 # Fresh registry (or first write over a legacy-format file — legacy
394 # entries reappear on that scope's next regenerate_active).
395 printf '# Lintel jobs registry (cross-repo)\n\n'
396 printf '> One line per open job; data lives in each repo'"'"'s .claude/runtime/jobs/.\n'
397 printf '> Generated by bin/_jobs.sh. Do not edit by hand.\n\n'
398 fi
399 local d f id workflow status current_step
400 for d in "$LINTEL_JOBS_DIR"/*/; do
401 [ -d "$d" ] || continue
402 f="$d/job.yaml"
403 [ -f "$f" ] || continue
404 id=$(grep '^job_id:' "$f" | head -1 | awk '{print $2}')
405 workflow=$(grep '^workflow:' "$f" | head -1 | awk '{print $2}')
406 status=$(grep '^status:' "$f" | head -1 | awk '{print $2}')
407 current_step=$(grep '^current_step:' "$f" | head -1 | awk '{print $2}')
408 printf -- '- `%s` · %s · %s · step:%s · %s %s\n' \
409 "$id" "$workflow" "$status" "$current_step" "$scope_tag" "$marker"
410 done
411 } > "$tmp"
