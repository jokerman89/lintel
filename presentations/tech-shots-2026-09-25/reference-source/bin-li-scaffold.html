167 # Pure-bash literal substitution — NOT sed (battletest K1: GNU sed's `s///e`

171 _render="$(cat "$SCAFFOLDING_SRC/CLAUDE.md.template")"
172 _render="${_render//\{\{REPO_NAME\}\}/$NAME}"
173 _render="${_render//\{\{PACK\}\}/$PACK}"
174 _render="${_render//\{\{MODE\}\}/$MODE}"
175 _render="${_render//\{\{VOICE_TIER\}\}/$VOICE_TIER}"
176 printf '%s\n' "$_render" > CLAUDE.md
